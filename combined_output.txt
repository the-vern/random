import marimo

__generated_with = "0.9.0"
app = marimo.App(width="medium")


@app.cell
def _():
    import marimo as mo
    from datetime import datetime
    return datetime, mo


@app.cell
def _(mo):
    # Shared state: the list of processing steps (each step is a dict)
    get_history, set_history = mo.state([])
    return get_history, set_history


@app.cell
def _(mo):
    mo.md(
        """
        # Processing History Tracker
        Log each processing step you apply, review the history, and undo or clear it.
        """
    )
    return


@app.cell
def _(mo):
    step_name = mo.ui.dropdown(
        options=["Load data", "Clean", "Filter", "Transform", "Normalize", "Aggregate", "Export", "Other"],
        value="Clean",
        label="Step type",
    )
    description = mo.ui.text(label="Description", placeholder="e.g. dropped rows with nulls", full_width=True)
    status = mo.ui.radio(options=["success", "warning", "failed"], value="success", label="Status", inline=True)
    add_button = mo.ui.run_button(label="Add step")
    undo_button = mo.ui.run_button(label="Undo last", kind="warn")
    clear_button = mo.ui.run_button(label="Clear all", kind="danger")

    mo.vstack(
        [
            mo.hstack([step_name, status], justify="start", gap=2),
            description,
            mo.hstack([add_button, undo_button, clear_button], justify="start"),
        ]
    )
    return add_button, clear_button, description, status, step_name, undo_button


@app.cell
def _(
    add_button,
    clear_button,
    datetime,
    description,
    set_history,
    status,
    step_name,
    undo_button,
):
    # Handle button clicks (each run_button is True only on the run triggered by its click)
    if add_button.value:
        set_history(
            lambda h: h
            + [
                {
                    "#": len(h) + 1,
                    "time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                    "step": step_name.value,
                    "description": description.value,
                    "status": status.value,
                }
            ]
        )
    elif undo_button.value:
        set_history(lambda h: h[:-1])
    elif clear_button.value:
        set_history([])
    return


@app.cell
def _(get_history, mo):
    history = get_history()
    if history:
        failed = sum(1 for s in history if s["status"] == "failed")
        warned = sum(1 for s in history if s["status"] == "warning")
        summary = mo.hstack(
            [
                mo.stat(value=len(history), label="Total steps"),
                mo.stat(value=warned, label="Warnings"),
                mo.stat(value=failed, label="Failed"),
            ],
            justify="start",
        )
        out = mo.vstack([summary, mo.ui.table(history, selection=None, pagination=False)])
    else:
        out = mo.callout(mo.md("No steps recorded yet. Add one above."), kind="neutral")
    out
    return


if __name__ == "__main__":
    app.run()

"""S3 bucket and object management helper built on boto3."""

import logging
import os
from typing import Dict, Iterator, List, Optional

import boto3
from botocore.exceptions import ClientError

logger = logging.getLogger(__name__)


class S3ManagerError(Exception):
    """Raised when an S3 operation fails."""


class S3Manager:
    """Wrapper around boto3 for common S3 bucket and object operations.

    Credentials are resolved via the standard boto3 chain (env vars,
    shared credentials file, IAM role) unless passed explicitly.
    """

    def __init__(
        self,
        region_name: Optional[str] = None,
        aws_access_key_id: Optional[str] = None,
        aws_secret_access_key: Optional[str] = None,
        aws_session_token: Optional[str] = None,
        profile_name: Optional[str] = None,
        endpoint_url: Optional[str] = None,
    ):
        session = boto3.session.Session(
            region_name=region_name,
            aws_access_key_id=aws_access_key_id,
            aws_secret_access_key=aws_secret_access_key,
            aws_session_token=aws_session_token,
            profile_name=profile_name,
        )
        self.region_name = session.region_name
        self.client = session.client("s3", endpoint_url=endpoint_url)
        self.resource = session.resource("s3", endpoint_url=endpoint_url)

    # ------------------------------------------------------------------
    # Bucket operations
    # ------------------------------------------------------------------
    def list_buckets(self) -> List[str]:
        """Return the names of all buckets owned by the account."""
        try:
            response = self.client.list_buckets()
            return [b["Name"] for b in response.get("Buckets", [])]
        except ClientError as e:
            raise S3ManagerError(f"Failed to list buckets: {e}") from e

    def bucket_exists(self, bucket_name: str) -> bool:
        """Check whether a bucket exists and is accessible."""
        try:
            self.client.head_bucket(Bucket=bucket_name)
            return True
        except ClientError as e:
            code = e.response["Error"]["Code"]
            if code in ("404", "NoSuchBucket", "NotFound"):
                return False
            if code in ("403", "AccessDenied"):
                logger.warning("Bucket %s exists but access is denied", bucket_name)
                return True
            raise S3ManagerError(f"Failed to check bucket {bucket_name}: {e}") from e

    def create_bucket(self, bucket_name: str, region: Optional[str] = None) -> None:
        """Create a bucket. Handles the us-east-1 LocationConstraint quirk."""
        region = region or self.region_name or "us-east-1"
        try:
            if region == "us-east-1":
                self.client.create_bucket(Bucket=bucket_name)
            else:
                self.client.create_bucket(
                    Bucket=bucket_name,
                    CreateBucketConfiguration={"LocationConstraint": region},
                )
            logger.info("Created bucket %s in %s", bucket_name, region)
        except ClientError as e:
            raise S3ManagerError(f"Failed to create bucket {bucket_name}: {e}") from e

    def delete_bucket(self, bucket_name: str, force: bool = False) -> None:
        """Delete a bucket. With force=True, empties it first (including versions)."""
        try:
            if force:
                self.empty_bucket(bucket_name)
            self.client.delete_bucket(Bucket=bucket_name)
            logger.info("Deleted bucket %s", bucket_name)
        except ClientError as e:
            raise S3ManagerError(f"Failed to delete bucket {bucket_name}: {e}") from e

    def empty_bucket(self, bucket_name: str) -> None:
        """Delete all objects (and object versions) in a bucket."""
        try:
            bucket = self.resource.Bucket(bucket_name)
            bucket.objects.all().delete()
            bucket.object_versions.all().delete()
        except ClientError as e:
            raise S3ManagerError(f"Failed to empty bucket {bucket_name}: {e}") from e

    # ------------------------------------------------------------------
    # Object operations
    # ------------------------------------------------------------------
    def upload_file(
        self,
        local_path: str,
        bucket_name: str,
        key: Optional[str] = None,
        extra_args: Optional[Dict] = None,
    ) -> str:
        """Upload a local file. Returns the object key used.

        extra_args example: {"ContentType": "image/png", "ACL": "private"}
        """
        if not os.path.isfile(local_path):
            raise FileNotFoundError(f"Local file not found: {local_path}")
        key = key or os.path.basename(local_path)
        try:
            self.client.upload_file(local_path, bucket_name, key, ExtraArgs=extra_args)
            logger.info("Uploaded %s to s3://%s/%s", local_path, bucket_name, key)
            return key
        except ClientError as e:
            raise S3ManagerError(f"Failed to upload {local_path}: {e}") from e

    def upload_bytes(
        self,
        data: bytes,
        bucket_name: str,
        key: str,
        content_type: Optional[str] = None,
    ) -> None:
        """Upload in-memory bytes as an object."""
        kwargs = {"Bucket": bucket_name, "Key": key, "Body": data}
        if content_type:
            kwargs["ContentType"] = content_type
        try:
            self.client.put_object(**kwargs)
        except ClientError as e:
            raise S3ManagerError(f"Failed to upload bytes to {key}: {e}") from e

    def download_file(
        self, bucket_name: str, key: str, local_path: Optional[str] = None
    ) -> str:
        """Download an object to disk. Returns the local path."""
        local_path = local_path or os.path.basename(key)
        parent = os.path.dirname(local_path)
        if parent:
            os.makedirs(parent, exist_ok=True)
        try:
            self.client.download_file(bucket_name, key, local_path)
            logger.info("Downloaded s3://%s/%s to %s", bucket_name, key, local_path)
            return local_path
        except ClientError as e:
            raise S3ManagerError(f"Failed to download {key}: {e}") from e

    def get_object_bytes(self, bucket_name: str, key: str) -> bytes:
        """Read an object's contents into memory."""
        try:
            response = self.client.get_object(Bucket=bucket_name, Key=key)
            return response["Body"].read()
        except ClientError as e:
            raise S3ManagerError(f"Failed to get {key}: {e}") from e

    def get_object_text(
        self, bucket_name: str, key: str, encoding: str = "utf-8"
    ) -> str:
        """Read an object's contents as text."""
        return self.get_object_bytes(bucket_name, key).decode(encoding)

    def delete_file(self, bucket_name: str, key: str) -> None:
        """Delete a single object."""
        try:
            self.client.delete_object(Bucket=bucket_name, Key=key)
            logger.info("Deleted s3://%s/%s", bucket_name, key)
        except ClientError as e:
            raise S3ManagerError(f"Failed to delete {key}: {e}") from e

    def delete_files(self, bucket_name: str, keys: List[str]) -> List[str]:
        """Delete many objects (batched, 1000 per request). Returns failed keys."""
        failed: List[str] = []
        try:
            for i in range(0, len(keys), 1000):
                batch = [{"Key": k} for k in keys[i : i + 1000]]
                response = self.client.delete_objects(
                    Bucket=bucket_name, Delete={"Objects": batch, "Quiet": True}
                )
                failed.extend(err["Key"] for err in response.get("Errors", []))
        except ClientError as e:
            raise S3ManagerError(f"Failed to delete objects: {e}") from e
        return failed

    def file_exists(self, bucket_name: str, key: str) -> bool:
        """Check whether an object exists."""
        try:
            self.client.head_object(Bucket=bucket_name, Key=key)
            return True
        except ClientError as e:
            if e.response["Error"]["Code"] in ("404", "NoSuchKey", "NotFound"):
                return False
            raise S3ManagerError(f"Failed to check {key}: {e}") from e

    def get_file_metadata(self, bucket_name: str, key: str) -> Dict:
        """Return size, content type, last modified, ETag and user metadata."""
        try:
            r = self.client.head_object(Bucket=bucket_name, Key=key)
            return {
                "size": r["ContentLength"],
                "content_type": r.get("ContentType"),
                "last_modified": r["LastModified"],
                "etag": r["ETag"].strip('"'),
                "metadata": r.get("Metadata", {}),
            }
        except ClientError as e:
            raise S3ManagerError(f"Failed to get metadata for {key}: {e}") from e

    def list_files(
        self, bucket_name: str, prefix: str = "", max_items: Optional[int] = None
    ) -> List[Dict]:
        """List objects under a prefix (paginated). Returns key/size/last_modified."""
        return list(self.iter_files(bucket_name, prefix, max_items))

    def iter_files(
        self, bucket_name: str, prefix: str = "", max_items: Optional[int] = None
    ) -> Iterator[Dict]:
        """Lazily iterate over objects under a prefix."""
        paginator = self.client.get_paginator("list_objects_v2")
        count = 0
        try:
            for page in paginator.paginate(Bucket=bucket_name, Prefix=prefix):
                for obj in page.get("Contents", []):
                    yield {
                        "key": obj["Key"],
                        "size": obj["Size"],
                        "last_modified": obj["LastModified"],
                    }
                    count += 1
                    if max_items and count >= max_items:
                        return
        except ClientError as e:
            raise S3ManagerError(f"Failed to list files in {bucket_name}: {e}") from e

    def copy_file(
        self,
        src_bucket: str,
        src_key: str,
        dest_bucket: str,
        dest_key: Optional[str] = None,
    ) -> None:
        """Copy an object within or across buckets."""
        dest_key = dest_key or src_key
        try:
            self.client.copy(
                {"Bucket": src_bucket, "Key": src_key}, dest_bucket, dest_key
            )
        except ClientError as e:
            raise S3ManagerError(f"Failed to copy {src_key}: {e}") from e

    def move_file(
        self,
        src_bucket: str,
        src_key: str,
        dest_bucket: str,
        dest_key: Optional[str] = None,
    ) -> None:
        """Move an object (copy then delete the source)."""
        self.copy_file(src_bucket, src_key, dest_bucket, dest_key)
        self.delete_file(src_bucket, src_key)

    def generate_presigned_url(
        self,
        bucket_name: str,
        key: str,
        expires_in: int = 3600,
        method: str = "get_object",
    ) -> str:
        """Create a temporary URL. method: 'get_object' or 'put_object'."""
        try:
            return self.client.generate_presigned_url(
                ClientMethod=method,
                Params={"Bucket": bucket_name, "Key": key},
                ExpiresIn=expires_in,
            )
        except ClientError as e:
            raise S3ManagerError(f"Failed to presign {key}: {e}") from e


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    s3 = S3Manager(region_name="us-east-1")

    bucket = "my-example-bucket-12345"
    if not s3.bucket_exists(bucket):
        s3.create_bucket(bucket)

    s3.upload_file("example.txt", bucket, "docs/example.txt")
    print(s3.list_files(bucket, prefix="docs/"))
    print(s3.get_object_text(bucket, "docs/example.txt"))
    print(s3.generate_presigned_url(bucket, "docs/example.txt", expires_in=600))
    s3.delete_file(bucket, "docs/example.txt")
